# Example 3: Functions with multiple parameters, strings, and while loops
# Demonstrates: function definitions with multiple parameters,
# return values, while loops, string literals, print

def greet(name):
    print name

def add(a, b):
    return a + b

def factorial(n):
    result = 1
    while n > 1:
        result = result * n
        n = n - 1
    return result

# Print greeting
greet("Hello, World!")

# Test multi-parameter function
sum = add(3, 7)
print sum

# Test factorial with while loop
f = factorial(5)
print f

# More arithmetic
pi = 3.14
radius = 5
area = pi * radius * radius
print area
